def clean_scroll(x):
    if pd.isna(x):
        return None
    s = str(x).replace('<font style="vertical-align: inherit;">', '').replace('</font>', '')
    try:
        return float(s)
    except ValueError:
        return None

by_marker = df.groupby('marker_id').agg(
    total=('assignment_id', 'count'),
    accepted=('status', lambda s: (s == 'ACCEPTED').sum()),
    rejected=('status', lambda s: (s == 'REJECTED').sum()),
    expired=('status', lambda s: (s == 'EXPIRED').sum()),

by_marker['accept_rate'] = (by_marker['accepted'] / by_marker['total']).round(3)
by_marker['fail_rate'] = ((by_marker['rejected'] + by_marker['expired']) / by_marker['total']).round(3)
